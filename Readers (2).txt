
@Test
public void testAssignStep() {
    OnboardingManager manager = new OnboardingManager();
    manager.addEmployee(new Employee(1, "Alice Johnson", 100));

    manager.addStep(new OnboardingStep(
            1, 1, "Submit Documents",
            StepStatus.PENDING, 10, 105));
    manager.addStep(new OnboardingStep(
            2, 1, "IT Setup",
            StepStatus.IN_PROGRESS, 20, 107));
    manager.addStep(new OnboardingStep(
            3, 1, "Manager Meeting",
            StepStatus.PENDING, 30, 110));

    // Successfully reassign a step to a new owner
    assertTrue(manager.assignStep(1, 99));
    assertEquals(99, manager.steps.get(0).ownerId);

    // Owner of other steps unchanged
    assertEquals(20, manager.steps.get(1).ownerId);
    assertEquals(30, manager.steps.get(2).ownerId);

    // Step does not exist
    assertFalse(manager.assignStep(999, 10));

    // Reassign to same owner is still valid
    assertTrue(manager.assignStep(2, 20));
    assertEquals(20, manager.steps.get(1).ownerId);
}

@Test
public void testGetReadySteps() {
    OnboardingManager manager = new OnboardingManager();
    manager.addEmployee(new Employee(1, "Alice Johnson", 100));
    manager.addEmployee(new Employee(2, "Bob Smith", 101));
    manager.addEmployee(new Employee(3, "Carol White", 102));
    manager.addEmployee(new Employee(4, "David Brown", 103));

    // Employee 1: steps 1 and 2 completed, step 3 is PENDING and ready
    // step 4 is PENDING but blocked by step 3 not being completed
    manager.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.COMPLETED, 10, 105));
    manager.addStep(new OnboardingStep(2, 1, "Background Check",
            StepStatus.COMPLETED, 10, 107));
    manager.addStep(new OnboardingStep(3, 1, "IT Setup",
            StepStatus.PENDING, 20, 110));
    manager.addStep(new OnboardingStep(4, 1, "Manager Meeting",
            StepStatus.PENDING, 30, 112));

    // Employee 2: step 5 is IN_PROGRESS - step 6 is not ready
    manager.addStep(new OnboardingStep(5, 2, "Submit Documents",
            StepStatus.IN_PROGRESS, 10, 105));
    manager.addStep(new OnboardingStep(6, 2, "IT Setup",
            StepStatus.PENDING, 20, 108));

    // Employee 3: all steps COMPLETED - nothing ready
    manager.addStep(new OnboardingStep(7, 3, "Submit Documents",
            StepStatus.COMPLETED, 10, 103));
    manager.addStep(new OnboardingStep(8, 3, "IT Setup",
            StepStatus.COMPLETED, 20, 106));

    // Employee 4: first step PENDING - only step 11 is ready (no prerequisites)
    // step 12 is blocked by step 11 not being completed
    manager.addStep(new OnboardingStep(11, 4, "Submit Documents",
            StepStatus.PENDING, 40, 108));
    manager.addStep(new OnboardingStep(12, 4, "IT Setup",
            StepStatus.PENDING, 40, 110));

    // Employee 1: step 3 is ready, step 4 is blocked by step 3
    List<OnboardingStep> result1 = manager.getReadySteps(1);
    assertEquals(1, result1.size());
    assertEquals(3, result1.get(0).stepId);

    // Employee 2: step 5 is IN_PROGRESS - step 6 not ready
    assertEquals(0, manager.getReadySteps(2).size());

    // Employee 3: all completed - nothing ready
    assertEquals(0, manager.getReadySteps(3).size());

    // Employee 4: step 11 is ready (no prerequisites), step 12 blocked by step 11
    List<OnboardingStep> result4 = manager.getReadySteps(4);
    assertEquals(1, result4.size());
    assertEquals(11, result4.get(0).stepId);

    // BLOCKED step blocks all subsequent steps regardless of their status
    OnboardingManager manager2 = new OnboardingManager();
    manager2.addEmployee(new Employee(5, "Eve Adams", 104));
    manager2.addStep(new OnboardingStep(13, 5, "Submit Documents",
            StepStatus.COMPLETED, 10, 105));
    manager2.addStep(new OnboardingStep(14, 5, "Background Check",
            StepStatus.BLOCKED, 10, 107));
    manager2.addStep(new OnboardingStep(15, 5, "IT Setup",
            StepStatus.IN_PROGRESS, 20, 110));
    assertEquals(0, manager2.getReadySteps(5).size());

    // No steps for employee - empty list
    OnboardingManager manager3 = new OnboardingManager();
    manager3.addEmployee(new Employee(6, "Frank Lee", 105));
    assertEquals(0, manager3.getReadySteps(6).size());

    // Steps added out of order - readiness must use stepId order not insertion order
    OnboardingManager manager4 = new OnboardingManager();
    manager4.addEmployee(new Employee(7, "Grace Kim", 106));
    manager4.addStep(new OnboardingStep(22, 7, "IT Setup",
            StepStatus.PENDING, 20, 110));
    manager4.addStep(new OnboardingStep(20, 7, "Submit Documents",
            StepStatus.COMPLETED, 10, 105));
    manager4.addStep(new OnboardingStep(21, 7, "Background Check",
            StepStatus.COMPLETED, 10, 107));
    List<OnboardingStep> result6 = manager4.getReadySteps(7);
    assertEquals(1, result6.size());
    assertEquals(22, result6.get(0).stepId);
}

@Test
public void testGetOwnerImpactReport() {
    OnboardingManager manager = new OnboardingManager();
    manager.addEmployee(new Employee(1, "Alice Johnson", 90));
    manager.addEmployee(new Employee(2, "Bob Smith", 91));
    manager.addEmployee(new Employee(3, "Carol White", 92));
    manager.addEmployee(new Employee(4, "David Brown", 93));
    manager.addEmployee(new Employee(5, "Eve Adams", 94));

    // Employee 1: step 1 COMPLETED, step 2 IN_PROGRESS (owner 20) -> owner 20 blocks emp 1
    manager.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.COMPLETED, 10, 100));
    manager.addStep(new OnboardingStep(2, 1, "Background Check",
            StepStatus.IN_PROGRESS, 20, 103));
    manager.addStep(new OnboardingStep(3, 1, "IT Setup",
            StepStatus.PENDING, 30, 107));

    // Employee 2: step 4 COMPLETED, step 5 BLOCKED (owner 20) -> owner 20 blocks emp 2
    manager.addStep(new OnboardingStep(4, 2, "Submit Documents",
            StepStatus.COMPLETED, 10, 100));
    manager.addStep(new OnboardingStep(5, 2, "Background Check",
            StepStatus.BLOCKED, 20, 103));
    manager.addStep(new OnboardingStep(6, 2, "IT Setup",
            StepStatus.PENDING, 30, 107));

    // Employee 3: step 7 IN_PROGRESS (owner 10) -> owner 10 blocks emp 3
    manager.addStep(new OnboardingStep(7, 3, "Submit Documents",
            StepStatus.IN_PROGRESS, 10, 103));
    manager.addStep(new OnboardingStep(8, 3, "IT Setup",
            StepStatus.PENDING, 30, 107));

    // Employee 4: step 9 PENDING (owner 30) -> PENDING first step, no owner is blocking
    manager.addStep(new OnboardingStep(9, 4, "Submit Documents",
            StepStatus.PENDING, 30, 103));
    manager.addStep(new OnboardingStep(10, 4, "IT Setup",
            StepStatus.PENDING, 20, 107));

    // Employee 5: all COMPLETED - no blocker
    manager.addStep(new OnboardingStep(11, 5, "Submit Documents",
            StepStatus.COMPLETED, 10, 100));
    manager.addStep(new OnboardingStep(12, 5, "IT Setup",
            StepStatus.COMPLETED, 20, 103));

    List<OwnerImpact> result = manager.getOwnerImpactReport();

    // owner 20 blocks emp 1 and emp 2 -> blockedCount=2
    // owner 10 blocks emp 3 -> blockedCount=1
    // owner 30 has only PENDING first steps -> not blocking
    assertEquals(2, result.size());
    assertEquals(20, result.get(0).ownerId);
    assertEquals(2, result.get(0).blockedCount);
    assertEquals(10, result.get(1).ownerId);
    assertEquals(1, result.get(1).blockedCount);

    // Naive-solution-breaker: count distinct employees not steps
    // Employee 2 has another non-completed step for owner 20,
    // but only the first non-completed step should count.
    OnboardingManager manager2 = new OnboardingManager();
    manager2.addEmployee(new Employee(1, "Alice Johnson", 90));
    manager2.addEmployee(new Employee(2, "Bob Smith", 91));
    manager2.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.IN_PROGRESS, 20, 103));
    manager2.addStep(new OnboardingStep(2, 2, "Submit Documents",
            StepStatus.IN_PROGRESS, 20, 103));
    manager2.addStep(new OnboardingStep(3, 2, "IT Setup",
            StepStatus.IN_PROGRESS, 20, 107));

    List<OwnerImpact> result2 = manager2.getOwnerImpactReport();
    // owner 20 blocks emp 1 (via step 1) and emp 2 (via step 2)
    // step 3 for emp 2 is not the first non-completed - step 2 is
    // so blockedCount = 2 distinct employees not 3 steps
    assertEquals(1, result2.size());
    assertEquals(20, result2.get(0).ownerId);
    assertEquals(2, result2.get(0).blockedCount);

    // Tie-break: two owners blocking same number - lower ownerId first
    OnboardingManager manager3 = new OnboardingManager();
    manager3.addEmployee(new Employee(1, "Alice Johnson", 90));
    manager3.addEmployee(new Employee(2, "Bob Smith", 91));
    manager3.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.IN_PROGRESS, 30, 103));
    manager3.addStep(new OnboardingStep(2, 2, "Submit Documents",
            StepStatus.IN_PROGRESS, 10, 103));

    List<OwnerImpact> result3 = manager3.getOwnerImpactReport();
    // owner 10 blocks emp 2, owner 30 blocks emp 1 - both count=1
    // tie-break: ownerId ascending -> owner 10 first
    assertEquals(10, result3.get(0).ownerId);
    assertEquals(30, result3.get(1).ownerId);

    // Steps added out of order - first non-completed must be based on stepId order
    OnboardingManager manager4 = new OnboardingManager();
    manager4.addEmployee(new Employee(6, "Frank Lee", 95));
    manager4.addStep(new OnboardingStep(22, 6, "IT Setup",
            StepStatus.IN_PROGRESS, 20, 110));
    manager4.addStep(new OnboardingStep(20, 6, "Submit Documents",
            StepStatus.COMPLETED, 10, 100));
    manager4.addStep(new OnboardingStep(21, 6, "Background Check",
            StepStatus.BLOCKED, 30, 105));

    List<OwnerImpact> result4 = manager4.getOwnerImpactReport();
    // stepId order: 20(COMP) -> 21(BLOCKED, owner 30) -> 22(IN_PROG, owner 20)
    // first non-completed is step 21 (BLOCKED, owner 30) -> owner 30 is the blocker
    assertEquals(1, result4.size());
    assertEquals(30, result4.get(0).ownerId);
    assertEquals(1, result4.get(0).blockedCount);

    // No blockers - all employees have PENDING first steps
    OnboardingManager manager5 = new OnboardingManager();
    manager5.addEmployee(new Employee(1, "Alice Johnson", 90));
    manager5.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.PENDING, 10, 103));
    assertTrue(manager5.getOwnerImpactReport().isEmpty());
}

@Test
public void testGetWorkloadSummary() {
    OnboardingManager manager = new OnboardingManager();
    manager.addEmployee(new Employee(1, "Alice Johnson", 100));
    manager.addEmployee(new Employee(2, "Bob Smith", 101));

    // Owner 10: 2 steps - one COMPLETED, one PENDING -> workload = 1
    manager.addStep(new OnboardingStep(1, 1, "Submit Documents",
            StepStatus.COMPLETED, 10, 105));
    manager.addStep(new OnboardingStep(2, 1, "IT Setup",
            StepStatus.PENDING, 10, 107));

    // Owner 20: 2 steps - both non-completed -> workload = 2
    manager.addStep(new OnboardingStep(3, 1, "Manager Meeting",
            StepStatus.IN_PROGRESS, 20, 110));
    manager.addStep(new OnboardingStep(4, 2, "Badge Setup",
            StepStatus.BLOCKED, 20, 108));

    // Owner 30: 1 step - COMPLETED -> should not appear
    manager.addStep(new OnboardingStep(5, 2, "HR Orientation",
            StepStatus.COMPLETED, 30, 106));

    Map<Integer, Integer> summary = manager.getWorkloadSummary();

    // Owner 10 has 1 non-completed step
    assertEquals(Integer.valueOf(1), summary.get(10));

    // Owner 20 has 2 non-completed steps
    assertEquals(Integer.valueOf(2), summary.get(20));

    // Owner 30 has no non-completed steps - must not appear
    assertFalse(summary.containsKey(30));

    // Only 2 owners in summary
    assertEquals(2, summary.size());

    // All steps completed - empty summary
    OnboardingManager manager2 = new OnboardingManager();
    manager2.addEmployee(new Employee(3, "Carol White", 102));
    manager2.addStep(new OnboardingStep(6, 3, "Submit Documents",
            StepStatus.COMPLETED, 10, 105));
    manager2.addStep(new OnboardingStep(7, 3, "IT Setup",
            StepStatus.COMPLETED, 20, 107));
    assertTrue(manager2.getWorkloadSummary().isEmpty());
}
